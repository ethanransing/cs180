    predicted_update = (1 - t) * predict_velocity(x_t, t, condition)
    ground_truth_update = clean_image - x_t

1.5 Iterative Denoising with Euler Sampling

@torch.inference_mode()
def euler_sample(x_start, t_start, step_size, velocity):
    x_end = x_start.clone()
    t = float(t_start)
    history = []
    while t < 1:
        clipped_step_size = min(step_size, 1.0 - t)
        x_end += clipped_step_size * velocity(x_end, t)
        t += clipped_step_size
        history.append((t, x_end.clone()))
    return x_end, history

2.1 Classifier-Free Guidance

def cfg_velocity(x, t, condition, w, predict_velocity, null_condition=''):
    conditional_v = predict_velocity(x, t, condition)
    null_v = predict_velocity(x, t, null_condition)
    return conditional_v + w * (conditional_v - null_v)

3.3 Visual Anagrams

def visual_anagrams(noise, condition_a, condition_b, predict_velocity, w=7, step_size=0.02):
    def anagram_velocity(img, t):
        v = cfg_velocity(img, t, condition_a, w, predict_velocity)
        flipped_v = rotate180(cfg_velocity(rotate180(img), t, condition_b, w, predict_velocity))
        return (v + flipped_v) / 2
    anagram, _ = euler_sample(noise, 0.0, step_size, anagram_velocity)
    return anagram

3.4 Hybrid Images

def make_hybrids(noise, condition_a, condition_b, predict_velocity, w=7, step_size=0.02):
    def hybrid_velocity(img, t):
        v_a = cfg_velocity(img, t, condition_a, w, predict_velocity)
        v_b = cfg_velocity(img, t, condition_b, w, predict_velocity)
        v_a_lowpass = gaussian_blur(v_a, [265, 265], [16.0, 16.0])
        v_b_lowpass = gaussian_blur(v_b, [265, 265], [16.0, 16.0])
        v_b_highpass = v_b - v_b_lowpass
        return v_a_lowpass + v_b_highpass
    hybrid, _ = euler_sample(noise, 0.0, step_size, hybrid_velocity)
    return hybrid

1.1 Implementing the UNet

class UnconditionalUNet(nn.Module):
    def __init__(self, in_channels: int, num_hiddens: int):
        super().__init__()
        self.conv1 = ConvBlock(in_channels, num_hiddens)
        self.down1 = DownBlock(num_hiddens, num_hiddens)
        self.down2 = DownBlock(num_hiddens, num_hiddens * 2)
        self.flatten = Flatten()
        self.unflatten = Unflatten(num_hiddens * 2)
        self.up1 = UpBlock(4 * num_hiddens, num_hiddens)
        self.up2 = UpBlock(2 * num_hiddens, num_hiddens)
        self.conv2 = ConvBlock(num_hiddens * 2, num_hiddens)
        self.output = nn.Conv2d(in_channels=num_hiddens, out_channels=1, kernel_size=3, stride=1, padding=1)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        x1 = self.conv1(x)
        x2 = self.down1(x1)
        x3 = self.down2(x2)
        z = self.unflatten(self.flatten(x3))
        z = self.up1(torch.cat([x3, z], dim=1))
        z = self.up2(torch.cat([x2, z], dim=1))
        z = self.conv2(torch.cat([x1, z], dim=1))
        return self.output(z)

def class_fm_forward(unet, x_1, c, p_uncond, num_ts):
    batch_size = x_1.shape[0]
    noise = torch.randn_like(x_1)
    t = torch.rand(batch_size, device=x_1.device)
    t_image = t.reshape(batch_size, 1, 1, 1)
    x_t = (1 - t_image) * noise + t_image * x_1
    mask = torch.rand(batch_size, device=x_1.device) >= p_uncond
    mask = mask.float()
    predicted_v = unet(x_t, c, t, mask)
    target_v = x_1 - noise
    return F.mse_loss(predicted_v, target_v)

2.6 Sampling from the Class-Conditioned UNet

@torch.inference_mode()
def class_fm_sample(unet, c, img_wh, num_ts, guidance_scale=5.0, seed=0):
    torch.manual_seed(seed)
    c = c.to(device)
    batch_size = c.shape[0]
    height, width = img_wh
    x = torch.randn(batch_size, 1, height, width, device=device)
    step_size = 1.0 / num_ts
    unconditional_mask = torch.zeros(batch_size, device=device)
    frames = [x.clone()]
    for step in range(num_ts):
        t = torch.full((batch_size,), step / num_ts, device=device)
        conditional_v = unet(x, c, t)
        unconditional_v = unet(x, c, t, unconditional_mask)
        velocity = unconditional_v + guidance_scale * (conditional_v - unconditional_v)
        x = x + step_size * velocity
        frames.append(x.clone())
    frames = torch.stack(frames, dim=1)
    return x, frames
